cmake_minimum_required(VERSION 3.24)
project(s20_pipeline LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The full native pipeline currently requires macOS with Metal")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
include(FetchContent)
set(KISS_SOURCE "" CACHE PATH "Optional local KISS-ICP cpp/kiss_icp source")
if(NOT KISS_SOURCE)
  FetchContent_Declare(kiss GIT_REPOSITORY https://github.com/PRBonn/kiss-icp.git GIT_TAG b16835283aee62f7d5e2bdf6c1c3bb2930de74ff SOURCE_SUBDIR cpp/kiss_icp)
  FetchContent_MakeAvailable(kiss)
else()
  add_subdirectory("${KISS_SOURCE}" kiss EXCLUDE_FROM_ALL)
endif()
foreach(target IN ITEMS reconstruct refine_poses)
  if(target STREQUAL "reconstruct")
    set(source main.cpp)
  else()
    set(source refine_poses.cpp)
  endif()
  add_executable(s20_${target} ${source})
  target_link_libraries(s20_${target} PRIVATE kiss_icp_core)
  target_compile_options(s20_${target} PRIVATE -O3 -Wall -Wextra)
endforeach()
foreach(target IN ITEMS geometry blend)
  add_executable(s20_${target} ${target}.mm)
  target_compile_options(s20_${target} PRIVATE -O3 -fobjc-arc)
  target_link_libraries(s20_${target} PRIVATE "-framework Foundation" "-framework Metal")
endforeach()
add_library(s20_collector SHARED collector.mm)
target_compile_options(s20_collector PRIVATE -O3 -fobjc-arc)
target_link_libraries(s20_collector PRIVATE "-framework Foundation" "-framework Metal")
